% 复选框定义
% 原始模板为 "□"（U+25A1），此处使用 amssymb 的 $\square$ 以确保跨字体兼容
% $\square$ 为空框，$\boxtimes$ 为带×框
\newcommand{\checkedbox}{\makebox[0pt][l]{\raisebox{0.12em}{\tiny$\boxtimes$}}$\square$}
\newcommand{\emptybox}{$\square$}

% 根据 \topicXxx 是否为空来显示复选框
\newcommand{\makecheckbox}[1]{%
    \ifx#1\empty
        \emptybox{}%
    \else
        \checkedbox{}%
    \fi
}
